# ============================================================
# 环境变量（启动前设置）
# ============================================================
export OMP_NUM_THREADS=3
export MKL_NUM_THREADS=3
export OPENBLAS_NUM_THREADS=3
export GOAI_ASYNC=1  # 可选：启用异步流水线

# ============================================================
# ① 构建数据集（流式分片，峰值内存低）
# ============================================================
python scripts/build_dataset.py \
  --src data/games/games/ \
  --out data/sgf_19x19_full.npz \
  --board-size 19 \
  --chunk-size 50000 \
  --tmp-dir data/tmp

# 验证 npz 包含 game_ids（必须输出 True）
python -c "import numpy as np; d=np.load('data/sgf_19x19_full.npz'); print('game_ids' in d)"

# ============================================================
# ② SFT 监督训练（4卡 NPU）
# ============================================================
# 推荐配置：大 Batch + 深度预取 + ONNX 导出
torchrun --nproc_per_node=4 scripts/train_sft.py \
  --data data/sgf_19x19_full.npz \
  --device npu --board-size 19 \
  --backbone-channels 192 --backbone-res-blocks 17 \
  --batch-size 3200 --epochs 2 \
  --lr 0.002 --weight-decay 0.0001 \
  --attention-mode mix --num-attention-layers 4 --num-heads 4 \
  --attn-mode window_global --attn-window 7 \
  --attention-dropout 0.1 --label-smoothing 0.1 \
  --gradient-accumulation-steps 1 \
  --use-amp --use-ema \
  --prefetch-workers 8 --prefetch-depth 8 \
  --log-every 50 --eval-every 500 --save-every 500 \
  --resume models/sft_19x19_v12.pth \
  --out models/sft_19x19_v15.pth \
  --export-onnx

# 备用：显存不足时减小 batch
# --batch-size 1600 --gradient-accumulation-steps 2

# ============================================================
# ③ RL 自对弈训练（1卡 NPU + 8进程 CPU）
# ============================================================
# 标准模式：8进程并行
python scripts/selfplay_train.py \
  --board-size 19 --iters 20 --sims 64 \
  --parallel-games 8 \
  --mcts-threads 3 \
  --onnx-model models/sft_19x19_v15.onnx \
  --batch-cap 64 \
  --expand-topk 16 \
  --buffer-size 500 \
  --use-rollout --leaf-ab-depth 2 \
  --c-puct 2.0 --virtual-loss 8.0 \
  --model models/sft_19x19_v15.pth \
  --out models/az_best.pth

# 异步模式：生成与训练完全并行（额外 1.5-2x 提速）
GOAI_ASYNC=1 python scripts/selfplay_train.py \
  --board-size 19 --iters 20 --sims 64 \
  --parallel-games 8 \
  --mcts-threads 3 \
  --onnx-model models/sft_19x19_v15.onnx \
  --batch-cap 64 \
  --games-per-iter 20 \
  --model models/sft_19x19_v15.pth \
  --out models/az_best.pth

# ============================================================
# ④ WebUI 人机对弈
# ============================================================
# 推荐配置：ONNX + 多线程 + 策略深度2
python scripts/webui.py --port 7860 --device cpu --priors-leaf \
    --onnx models/sft_19x19_v15.onnx \
    --mode hybrid --hybrid-sims 16 --hybrid-blend 0.5 \
    --expand-topk 16 --num-threads 4 \
    --policy-depth 2 --policy-width 4 --policy-topk 12

# 高质量模式：纯 MCTS（较慢但更强）
python scripts/webui.py --port 7860 --device cpu --priors-leaf \
    --onnx models/sft_19x19_v15.onnx \
    --mode mcts --simulations 64 --num-threads 4

# ============================================================
# ⑤ 评估棋力
# ============================================================
python scripts/evaluate.py \
  --model models/sft_19x19_v15.pth \
  --board-size 19 --device npu \
  --mode random --num-games 100 \
  --use-mcts --simulations 400 --num-threads 4

python scripts/eval_elo.py \
  --model models/sft_19x19_v15.pth \
  --board-size 19 --device npu \
  --num-games 50 --simulations 400

# ============================================================
# 关键优化总结
# ============================================================
# 1. SGF 元数据加权：解析 PB/PW 等级，高质量棋谱更高权重
# 2. 手数比例软标签 (D)：tanh(score × alpha)，连续胜率
# 3. MSE Loss：使用 winrates 时自动切换
# 4. ConvNeXt 风格：5x5 深度卷积 + LayerNorm + GELU (--arch convnext)
# 5. 动态 topk：早期8/中期16/后期32 (自动启用)
# 6. 动态 virtual_loss：早期2/中期6/后期12 (自动启用)
# 7. 策略剪枝：跳过 prior < 0.01 的候选
# 8. 多样化 rollout：4 种温度策略轮换 (--use-diverse-rollout)
# 9. ONNX 后端：CPU 推理 3-5x 提速
# 10. 进程级并行：8进程绕过 GIL
# 11. 异步流水线：生成与训练完全并行
