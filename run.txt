# ============================================================
# ⓪ 推荐训练参数（与 run.py 内置配置一致）
# ============================================================
# 下表是"当前推荐"，与 run.py 的 default_argv 保持一致（run.py 是代码入口，
# 以它为准）。下方 ②③④ 的方案 A/B/C 是不同硬件配置的变体，不是推荐值。
#
# 【硬件前提，务必先看】
#   · 910A 无 BF16  → NPU 一律 FP16 + GradScaler（代码已自动处理，勿手改）
#   · NPU 上 torch.compile(inductor) 不可用 → SFT 的 --compile 对 NPU 无效
#   · NPU 注意力走手写 math（sdpa_force_math=True），CANN 的 Flash 后端不稳
#   · NPU 的 pin_memory 关闭 → 双缓冲 H2D 只在 CUDA 生效
#
# 【SFT 推荐（单卡 A100 40G，BF16 + flash-attn）= 下方方案 B】
#   以下即 A100 40G 上实测可跑的配置（--batch-size 3500 实测通过）：
#   --batch-size 3500          与 --lr 0.00421 配套（由 2500/0.00356 按平方根
#                              缩放律换算：0.00356×√(3500/2500)=0.00421）。
#                              显存不足再降 batch 时须同步降 lr，或用
#                              --gradient-accumulation-steps 补回等效 batch
#   --lr 0.00421               勿单独改 batch 或 lr
#   --value-res-blocks 8       ⚠ ValueNetwork 的 ResBlock **不受
#                              --use-checkpoint 保护**，是最大一块常驻激活，
#                              11→8 是降显存的关键之一
#   --compile 1（不加 --compile-mode）  默认 default 模式。**不要**用
#                              reduce-overhead：它走 CUDA Graphs，会维持一个
#                              不归还的私有内存池，临界 batch 下直接 OOM
#   --attn-mode window_global --attn-window 7
#   ⚠ 降显存不要动 --attn-window（要往**大**调，详见方案 B 的说明）
#   --use-amp 1 --use-ema 1 --use-checkpoint 1 --flash-attn 1
#   --prefetch-workers 32 --prefetch-depth 32    数据预取
#   --eval-every 2000 --save-every 500           省时间与磁盘
#   --swanlab-every 10         SwanLab 上报频率与 --log-every 解耦：stdout 仍
#                              每 50 步一行（不刷屏），曲线每 10 步一个点。
#                              默认 0 = 跟随 --log-every（旧行为）
#   注意：方案 A（4 卡 NPU）与方案 C（4 卡 A100）仍是早期 batch 较大的旧配置，
#         换硬件前请按 ⓪ 节的显存结论重新标定。
#
# 【RL 推荐（1 卡 910A 训练 + 8 进程 CPU 搜索）】
#   标注 [默认] 的即代码默认值，不传也是这个值；标注 [刻意下调] 的需要显式传，
#   是为 19 路 + 8 进程 CPU 预算主动压低的（改了会显著变慢）。
#   --td 1 [默认]                TD C+E 价值标签；--td 0 为旧 tanh 基线
#   --batch-size 256 [默认]      NPU 甜点（显存约 2-3x 旧 64）
#   --mcts-vector-backup 1 [默认] 回传 numpy 批量（数值等价原实现）
#   --grad-accum-steps 1 [默认]  保持现状；开 4 需同时 --lr 0.0015~0.002
#   --sims 48 [刻意下调]         默认 400。19 路每步模拟数，直接用默认会慢到
#                               不可用；48 是 8 进程 CPU 上的可用档
#   --expand-topk 16 [刻意下调]  默认 64。展开候选截断，16 已足够且省 4x 前向
#   --parallel-games 8 --mcts-threads 3           24 核：8×3≈24
#   ⚠ --spec-prefetch 1 是默认值，但只有 --mcts-threads>=4 时才真正生效；
#     当前 3 线程下不启用，想用要调到 4 以上（注意会挤占 CPU 给 8 进程）
#   ⚠ 不要传 --onnx-model：当前实现把 .onnx 路径当 torch 权重 torch.load，
#     而 ONNX 是 protobuf 格式，启动即抛 UnpicklingError（已实测）。
#     自对弈推理目前与训练共用同一个 torch 模型（MCTS 推理也在 NPU 上跑），
#     功能可用但不是 CPU/ONNX 加速路径。
#
# 【验证状态】
#   已在本机（CPU）9x9 冒烟验证：TD 0/1、grad-accum 1/4、vector-backup 0/1
#   全部跑通且 loss 正常收敛；mcts-vector-backup 0/1 结果逐位一致。
#   未在 910A/A100 实测：所有耗时与吞吐数字（见文末"性能基准"）均为估算。

# 方案 B: 单卡 A100 40G (当前实跑配置)
# pip install flash-attn --no-build-isolation
#
# 【为什么以前 OOM、现在能到 3500】根因不是 batch 本身，而是
#   --compile-mode reduce-overhead：它走 CUDA Graphs，会维持一个**不归还**的
#   私有内存池，在临界 batch 上直接把显存吃光。去掉该参数（只留 --compile 1）
#   后，40G 卡上 3500 实测通过。同期另外两处调整：
#     · --value-res-blocks 11 → 8
#       ValueNetwork 的 ResBlock **不受 --use-checkpoint 保护**
#       （backbone 有 checkpoint，value head 没有），是最大一块常驻激活。
#     · --attn-window 5 → 7（见下）
#
# ⚠ 不要靠调小 --attn-window 省显存：window_global 的显存正比于
#   nW×(ws²+ng)，而 ng=ceil(19/ws)² 在 ws 变小时暴涨。实测相对倍数：
#   ws=3 → 5.4× 更差 | ws=5 → 1.26× 更差 | ws=7 → 基准 | ws=11 → 略好
#   所以 ws 要往**大**调（ws=11 或 19），不是往小调。
#
# 【SwanLab】上报频率与 stdout 解耦：--log-every 50 控制 stdout（不刷屏），
#   --swanlab-every 10 控制曲线密度（每 epoch 约 958 点而非 192 点）。
#   认证：平台只能传 `--swanlab-api-key <值>`，**环境变量不可用**。
#   切勿把真实 key 写进本文件或贴到聊天里；贴出过就去 swanlab.cn 吊销重发。
torchrun --nproc_per_node=1 scripts/train_sft.py \

  --device cuda --board-size 19 \
  --backbone-channels 192 --backbone-res-blocks 17 \
  --res-blocks 8 --convnext-blocks 4 --attn-blocks 5 \
  --value-channels 96 --value-res-blocks 8 \
  --policy-channels 128 --policy-layers 3 \
  --batch-size 3500 --epochs 1 \
  --lr 0.00421 --weight-decay 0.0001 \
  --attention-mode mix --num-attention-layers 4 --num-heads 4 \
  --attn-mode window_global --attn-window 7 \
  --attention-dropout 0.1 --label-smoothing 0.1 \
  --gradient-accumulation-steps 1 \
  --use-amp 1 --use-ema 1 \
  --compile 1 \
  --flash-attn 1 \
  --prefetch-workers 32 --prefetch-depth 32 \
  --log-every 50 --swanlab-every 10 --eval-every 2000 --save-every 500 \
  --early-stop 1 --early-stop-patience 3 \
  --out models/sft_19x19_v18.pth \
  --export-onnx 1 --use-checkpoint 1 \
  --swanlab 1 --ver v18

# ③' RL 完整参数表（selfplay_train.py，53 项；默认值均取自 --help/源码）
# ============================================================
# 用法：完整命令加 --help；下表为默认值，命令行显式传入即覆盖。
#
# --- 初始权重 / 对局规模 ---
#   --model PATH            初始权重（如 SFT 预训练）；默认 None
#   --board-size N          棋盘大小；默认 9
#   --iters N               迭代轮数；默认 5
#   --games N               每轮自对弈局数（串行模式）；默认 4
#   --sims N                每步 MCTS 模拟数；默认 400
#   --max-moves N           单局手数上限；默认 None = 3×点数
#   --temperature F         初始采样温度；默认 1.0
#   --buffer-size N         replay buffer 容量（局数，非样本数）；默认 500
#
# --- 训练优化 ---
#   --batch-size N          训练 batch（NPU 甜点）；默认 256
#   --epochs N              每轮训练遍数；默认 2
#   --lr F                  学习率；默认 1e-3
#   --weight-decay F        权重衰减；默认 1e-4
#   --value-lr-mult F       value head 学习率倍率；默认 0.5
#   --clip-grad F           梯度裁剪范数（0=关闭）；默认 1.0
#   --grad-accum-steps N    梯度累积 micro-batch 数；默认 1（>1 时等效
#                           batch×N、step÷N，建议 --lr 调高 1.4~2 倍）
#   --use-ema {0,1}         EMA 权重；默认 0
#   --no-augment {0,1}      关闭 8 对称增强；默认 0（开启）
#
# --- MCTS 搜索质量 ---
#   --expand-topk N         展开候选截断（CPU 推荐 32-64）；默认 64
#   --expand-chunk N        展开期 α-β 界截断块（0=关闭）；默认 0
#   --c-puct F              PUCT 探索系数；默认 2.0
#   --virtual-loss F        虚拟损失系数；默认 8.0
#   --num-threads N         MCTS 多线程数（进程内）；默认 8
#   --leaf-ab-depth N       叶内 α-β 深度；默认 2
#   --policy-pruning-thresh F  跳过 prior < 阈值的候选；默认 0.01
#   --dynamic-topk {0,1}    动态 topk（早8/中16/晚32）；默认 0
#   --dynamic-virtual-loss {0,1}  动态 virtual loss（2/6/12）；默认 0
#
# --- 并行 / 吞吐 ---
#   --parallel-games N      并行自对弈进程数（建议 cores/3，24核→8）；默认 8
#   --mcts-threads N        每进程 MCTS 线程（建议 cores/parallel_games）；默认 3
#   --onnx-model PATH       ONNX 权重（CPU 推理 3-5x）；默认 None
#   --batch-cap N           MCTS 批量展开上限；默认 64
#   --result-queue-max N    结果队列容量；默认 100
#   --mcts-vector-backup {0,1}  回传 visit/value_sum 走 numpy 批量；默认 1
#   --spec-prefetch {0,1}   worker 推测预评估（num_threads>=4 才生效）；默认 1
#
# --- Rollout ---
#   --use-rollout {0,1}     LightPLS rollout；默认 0
#   --rollout-lambda F      rollout 融合权重；默认 0.25
#   --rollout-steps N       rollout 最大步数；默认 60
#   --use-diverse-rollout {0,1}  4 种温度轮换；默认 0
#
# --- 异步流水线（生成与训练并行）---
#   --async-pipeline {0,1}  启用异步（须配 --games-per-iter）；默认 0
#   --games-per-iter N      异步模式每轮生成局数；默认 10
#
# --- TD 价值标签（C+E 混合）---
#   --td {0,1}              TD 标签（0=旧 tanh 软化基线）；默认 1
#   --td-steps N            n-step 前看步数（数据下标空间）；默认 3
#   --td-alpha-init F       α 调度初值（开局偏 r_soft）；默认 0.2
#   --td-alpha-end F        α 调度终值（残局偏 v_td）；默认 0.9
#
# --- 设备 / 分布式 ---
#   --device DEV            auto/cuda/npu/cpu；默认 auto

# ============================================================
# 性能基准与预期效果
# ============================================================
# 测试环境: 24核 CPU + 4卡 NPU 910A + 64GB RAM
#
# V17 架构参数:
#   - Backbone: 11.28M (ResBlock 8 + ConvNeXt 4 + Attention 5)
#   - Value:    1.01M (96ch, 11 blocks)
#   - Policy:   173k (128ch, 3 layers)
#   - Total:    12.46M (vs V12 12.67M, 基本持平)
#
# V12 vs V16 vs V17 对比:
#   V12: 12.67M (back=12.40M, val=0.26M, pol=0.006M) -> top1=50%
#   V16: 9.27M  (back=8.32M, val=0.93M, pol=0.016M) -> top1=40%
#   V17: 12.46M (back=11.28M, val=1.01M, pol=0.17M) -> 预期 >50%
#

# ============================================================
# 关键优化总结
# ============================================================
# 1. SGF 元数据加权：解析 PB/PW 等级，高质量棋谱更高权重
# 2. 手数比例软标签 (D)：tanh(score × alpha)，连续胜率
# 3. MSE Loss：使用 winrates 时自动切换
# 4. ConvNeXt 风格：5x5 深度卷积 + LayerNorm + GELU (--arch convnext)
# 5. 混合架构 Backbone：ResBlock (局部) + ConvNeXt (感受野) + Attention (全局)
# 6. 动态 topk：早期8/中期16/后期32 (自动启用)
# 7. 动态 virtual_loss：早期2/中期6/后期12 (自动启用)
# 8. 策略剪枝：跳过 prior < 0.01 的候选
# 9. 多样化 rollout：4 种温度策略轮换 (--use-diverse-rollout)
# 10. ONNX 后端：CPU 推理 3-5x 提速
# 11. 进程级并行：8进程绕过 GIL
# 12. 异步流水线：生成与训练完全并行 (--async-pipeline)
# 13. SwanLab 实验跟踪：实时可视化训练指标 (--swanlab)
# 14. 梯度累积 (--grad-accum-steps)：摊薄优化器步开销；配 --lr 1.4~2 倍
# 15. 双缓冲 H2D（CUDA）：pinned 槽交替 + 每槽 CUDA Event，搬运与计算重叠
# 16. MCTS 回传向量化 (--mcts-vector-backup，默认 1)：visit/value_sum 走 numpy 批量
# 17. 落子虚影（WebUI）：鼠标悬停显示半透明棋子，判定与点击共用 hitTest
