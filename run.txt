export OPENBLAS_NUM_THREADS=3

# ============================================================
# ① 构建数据集（流式分片，峰值内存低）
# ============================================================
python scripts/build_dataset.py \
  --src data/games/games/ \
  --out data/sgf_19x19_full.npz \
  --board-size 19 \
  --chunk-size 50000 \
  --tmp-dir data/tmp

# 验证 npz 包含 game_ids（必须输出 True）
python -c "import numpy as np; d=np.load('data/sgf_19x19_full.npz'); print('game_ids' in d)"

# ============================================================
# ② SFT 监督训练
# ============================================================
# 方案 A: 4卡 NPU (当前环境)
# 注意: --swanlab 需已登录 swanlab.cn，训练结束后云端查看图表
torchrun --nproc_per_node=4 scripts/train_sft.py \
  --data data/sgf_19x19_full.npz \
  --device npu --board-size 19 \
  --backbone-channels 192 --backbone-res-blocks 17 \
  --res-blocks 8 --convnext-blocks 4 --attn-blocks 5 \
  --value-channels 96 --value-res-blocks 11 \
  --policy-channels 128 --policy-layers 3 \
  --batch-size 3200 --epochs 1 \
  --lr 0.002 --weight-decay 0.0001 \
  --attention-mode mix --num-attention-layers 4 --num-heads 4 \
  --attn-mode window_global --attn-window 5 \
  --attention-dropout 0.1 --label-smoothing 0.1 \
  --gradient-accumulation-steps 1 \
  --use-amp 1 --use-ema 1 \
  --prefetch-workers 16 --prefetch-depth 16 \
  --log-every 50 --eval-every 500 --save-every 500 \
  --early-stop 1 --early-stop-patience 3 \
  --out models/sft_19x19_v17.pth \
  --export-onnx 1 --use-checkpoint 1 \
  --swanlab 1 --project go-ai --name sft_v17_npu4

# ============================================================
# ③ RL 自对弈训练（1卡 NPU + 8进程 CPU）
# ============================================================
# 标准模式：8进程并行
python scripts/selfplay_train.py \
  --board-size 19 --iters 20 --sims 48 \
  --parallel-games 8 \
  --mcts-threads 3 \
  --onnx-model models/sft_19x19_v17.onnx \
  --batch-cap 64 \
  --expand-topk 16 \
  --buffer-size 500 \
  --use-rollout 1 --leaf-ab-depth 2 \
  --c-puct 2.0 --virtual-loss 8.0 \
  --model models/sft_19x19_v17.pth \
  --out models/az_best.pth

# 异步模式：生成与训练完全并行（额外 1.5-2x 提速）
python scripts/selfplay_train.py \
  --async-pipeline 1 \
  --board-size 19 --iters 20 --sims 48 \
  --parallel-games 8 \
  --mcts-threads 3 \
  --onnx-model models/sft_19x19_v17.onnx \
  --batch-cap 64 \
  --games-per-iter 20 \
  --model models/sft_19x19_v17.pth \
  --out models/az_best.pth

# ============================================================
# 性能基准与预期效果
# ============================================================
# 测试环境: 24核 CPU + 4卡 NPU 910A + 64GB RAM
#
# V17 架构参数:
#   - Backbone: 11.28M (ResBlock 8 + ConvNeXt 4 + Attention 5)
#   - Value:    1.01M (96ch, 11 blocks)
#   - Policy:   173k (128ch, 3 layers)
#   - Total:    12.46M (vs V12 12.67M, 基本持平)
#
# V12 vs V16 vs V17 对比:
#   V12: 12.67M (back=12.40M, val=0.26M, pol=0.006M) -> top1=50%
#   V16: 9.27M  (back=8.32M, val=0.93M, pol=0.016M) -> top1=40%
#   V17: 12.46M (back=11.28M, val=1.01M, pol=0.17M) -> 预期 >50%
#
# SFT 训练:
#   - batch=3200, 4卡 NPU
#   - 预计: ~35-45min/epoch
#   - 速度: ~750 samples/s
#
# MCTS 推理 (CPU):
#   - sims=32, threads=4, topk=8: ~1-2s
#   - ONNX 加速 3-5x
#
# Self-play 生成 (8进程):
#   - sims=48, 8进程 × 3线程
#   - 预计: ~15-20s/局
#

#   - WebUI 响应: <3s

#   - 端到端训练周期: 缩短 10-20x

# ============================================================
# 关键优化总结
# ============================================================
# 1. SGF 元数据加权：解析 PB/PW 等级，高质量棋谱更高权重
# 2. 手数比例软标签 (D)：tanh(score × alpha)，连续胜率
# 3. MSE Loss：使用 winrates 时自动切换
# 4. ConvNeXt 风格：5x5 深度卷积 + LayerNorm + GELU (--arch convnext)
# 5. 混合架构 Backbone：ResBlock (局部) + ConvNeXt (感受野) + Attention (全局)
# 6. 动态 topk：早期8/中期16/后期32 (自动启用)
# 7. 动态 virtual_loss：早期2/中期6/后期12 (自动启用)
# 8. 策略剪枝：跳过 prior < 0.01 的候选
# 9. 多样化 rollout：4 种温度策略轮换 (--use-diverse-rollout)
# 10. ONNX 后端：CPU 推理 3-5x 提速
# 11. 进程级并行：8进程绕过 GIL
# 12. 异步流水线：生成与训练完全并行 (--async-pipeline)
# 13. SwanLab 实验跟踪：实时可视化训练指标 (--swanlab)
