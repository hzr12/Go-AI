# ---------- 0. 环境 ----------
# 硬件前提：910A 无 BF16 → 代码自动走 fp16+GradScaler；NPU 无 inductor → --compile 无效。
pip install -r requirements.txt
pip install scipy                  # requirements.txt 漏列，src/data/feature_v7.py 直接 import 它
pip install onnx onnxruntime       # 可选：ONNX 导出与 CPU 推理

# ---------- 1. 数据（已就位，⛔ 绝对不要重建）----------
# data/sgf_19x19_full.npz = 34,202,713 行 / 162,298 局 / 10 列（磁盘 447 MB，解压 12.3 GB）。
# ⛔ 不可 rebuild：build_dataset.py:314-316 有 game_id 复用 bug，重建会毁掉所有 join 链。
# ⛔ 不要整列读：np.load(f)['boards'] 会一次解压 12.3 GB（内存 <13 GB 必 OOM）→ 走 materialize + mmap。
# 真要从零构建（会毁 join 链，仅在完全没有 npz 时）：
#   python scripts/build_dataset.py --src data --out data/sgf_19x19_full.npz --board-size 19 --chunk-size 50000

# ---------- 2. 局级 sidecar games.npz ----------
# 产出 data/labels/games.npz（4 键 g_komi/g_score/g_rules/g_resign_side，约 1.2 MB）。
# 规模：扫 169,878 个 SGF + 落 .npy（152 s）→ 墙钟几十分钟；末行必打印匹配率报告。
#   小规模先验：加 --limit-games 200 --out tmp/games_probe.npz；已有 tmp/materialized/ 可加 --reuse-scan。
python scripts/build_games_sidecar.py --dataset data/sgf_19x19_full.npz --out data/labels/games.npz

# ---------- 3. KataGo 软标签（段 2 蒸馏用）----------
# 产出 data/labels/kata_labels.npz。--exe/--model/--config 自动从 katago/ 下递归查找。
# ⚠ --limit 0 才是「全部」；默认 500 会**只标 500 个位置**（不是打印上限，是硬截断）。
python scripts/label_sgf.py --game-frac 0.015 --move-lo 100 --move-hi 200 --limit 0 --out data/labels/kata_labels.npz
# join 到主数据集行号 → data/labels/soft_index.npz（命中行 × 362 策略）
python scripts/build_soft_index.py --data data/sgf_19x19_full.npz --labels data/labels/kata_labels.npz --out data/labels/soft_index.npz
# ⚠ 段 2 必须配 --soft-only-sampling（否则 mask=0 的行贡献恰好 0，policy 项被缩小约 100×）。

# ---------- 4. 训练 ----------
# 4a SFT：建网只看 scripts/train_sft.py 的 KATAGO_SE_CFG（12 通道 / 9,112,005 参数）；
#     那些 --backbone-channels / --attn-* 结构 flag **已归档，不参与建网**。显存未实测，先跑单卡。
python run.py --sh shell/train_sft_npu_1card.sh          # 单卡冒烟，建议先跑这个
python run.py --sh shell/train_sft_npu_4card_katago_se.sh # 4 卡正式；每卡 batch 别上调
# 4b V7（22 通道，--v7）：端到端冒烟吃 stdata 真实目标，40 步 / batch 8，CPU ~1 分钟。
python scripts/smoke_train_v7.py --files 40 --rows 256 --batch 8 --steps 40 --device cpu

# ---------- 5. 门禁（~3.5min）----------
# ⚠ 必须先钉线程，否则 -n auto 的 worker 抢核比串行还慢（test_max_batch 10.8s → 115s）。
#   setx OMP_NUM_THREADS 1（或每条命令前临时 set）
# ⚠ pytest.ini 的 addopts 已带 -n auto，别再手动加 -n。慢的回归用 @pytest.mark.slow 标记，
#   默认仍会跑（-m 只在命令行显式给）；当前没有值得单独跳过的，-m "not slow" 与全门禁同速。
python -m pytest tests -q
python -m pytest tests -q --durations=10   # 查最慢的 10 个
# ⚠ 加 CLI flag 后必须登记 tests/test_huber_loss.py 的冻结集，否则门禁自己先红。

# ---------- 6. 推理 / 自战 ----------

python scripts/cli_play.py --model models/sft_19x19_v12.pth --device cpu --simulations 400 --num-threads 8
python run.py --sh shell/train_rl_npu_1card.sh --model models/sft_19x19_v12.pth
