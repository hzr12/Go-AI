# ============================================================
# Go-AI 4卡 910A 全流程命令
# ============================================================

# ---- ① 构建数据集（流式分片，峰值内存低）----
python scripts/build_dataset.py \
  --src data/games/games/ \
  --out data/sgf_19x19_full.npz \
  --board-size 19 \
  --chunk-size 50000 \
  --tmp-dir data/tmp

# ---- ② 验证 npz 包含 game_ids（必须输出 True）----
python -c "import numpy as np; d=np.load('data/sgf_19x19_full.npz'); print('game_ids' in d)"

# ---- ③ 4卡 NPU 训练 ----
torchrun --nproc_per_node=4 scripts/train_sft.py \
  --data data/sgf_19x19_full.npz \
  --device npu --board-size 19 \
  --backbone-channels 192 --backbone-res-blocks 17 \
  --batch-size 400 --epochs 8 \
  --lr 0.002 --weight-decay 0.0001 \
  --attention-mode mix --num-attention-layers 4 --num-heads 4 \
  --attn-mode window --attn-window 7 \
  --attention-dropout 0.1 --label-smoothing 0.1 \
  --gradient-accumulation-steps 1 \
  --use-amp --use-ema \
  --log-every 50 --eval-every 500 --save-every 500 \
  --out models/sft_19x19_v6.pth

# ---- ④ 评估棋力（vs 随机）----
python scripts/evaluate.py \
  --model models/sft_19x19_v6.pth \
  --board-size 19 --device npu \
  --mode random --num-games 100 \
  --use-mcts --simulations 400 --num-threads 4

# ---- ⑤ ELO 评分 ----
python scripts/eval_elo.py \
  --model models/sft_19x19_v6.pth \
  --board-size 19 --device npu \
  --num-games 50 --simulations 400

# ---- ⑥ 自对弈测试 ----
python src/inference.py \
  --model models/sft_19x19_v6.pth \
  --board-size 19 --device npu --use-amp \
  --attn-mode window --attn-window 7 \
  --mode selfplay --games 10 \
  --use-mcts --simulations 400 --num-threads 4

# ---- WebUI ----
python scripts/webui.py --port 7860 --device cpu --priors-leaf --mode hybrid --hybrid-sims 24 --hybrid-blend 0.5 --expand-topk 12 --expand-chunk 8 --leaf-ab-depth 2 --policy-depth 3 --num-threads 1 --onnx --policy-topk 12 --policy-width 6
