# ---------- 4. 训练 ----------
# 4a SFT：建网只看 scripts/train_sft.py 的 KATAGO_SE_CFG（12 通道 / 9,112,005 参数）；

python scripts/smoke_train_v7.py --files 40 --rows 256 --batch 8 --steps 40 --device cpu

# ---------- 4c. 全流程训练参数（2×910A / 4×910A，SFT + RL）----------
# 【通则】
# ⚠ 建网只看一个源：12ch = KATAGO_SE_CFG，V7 22ch = katago_v7.py。
#   --backbone-channels / --res-blocks / --attn-* 等结构 flag 已归档，加了会被 run.py 忽略。
# ⚠ 910A 无 BF16 → 自动走 fp16+GradScaler，故 --scaler-init-scale 必须显式给（默认 65536 会炸）。
# ⚠ DDP 每 rank 复制参数 → **只能加 batch，不能加卡数**（卡多只是复制再去拓）。
# ⚠ 预取 worker 是**每 rank**的：2 卡 ×12 = 24，4 卡 ×6 = 24（24 核云端 CPU）。
#   ⚠ 且不钉线程会互相抢核，比串行还慢（实测 test_max_batch 10.8s → 115s）：
#      export OMP_NUM_THREADS=1 MKL_NUM_THREADS=1
# ⚠ LR 定标律（两个既有脚本都成立）：LR = 0.00356 × √(有效batch / 2500)
# ⚠ batch 对特征速度几乎无影响（32→1024 仅差 1.7%）⇒ batch 按显存选，别为特征保留。
#   V7 尤其如此：22 通道特征耗时 99.8% 在 iterLadders，但 CPU 侧与 batch 无关。
# ⚠ 段 1（V7）会**跳过** eval / early-stop / ONNX 导出 ⇒ 这三个 flag 传了也无效。

# ============ 段 1：SFT 从零训（22 通道 V7，policy/π_opp/value/futurepos）============
# 参数量 5,561,832（主干 5.48M）。score 系 8 项权重为 0（81% 的 SGF 是认输）。
# 显存实测（4 卡 32 GiB）：batch 2000/卡 在 27.6 GiB 时 OOM
#   （20.1 起始分配 + 6.93 碎片 + 4.36 CANN/HCCL）⇒ 4 卡下调到 1000。

# --- 2 × 910A：每卡 batch 2800，accum 1 ⇒ 有效 5600，LR 0.00533 ---
export OMP_NUM_THREADS=1 MKL_NUM_THREADS=1
python run.py --sh shell/train_sft_npu_2card.sh --v7 1 \
  --batch-size 2800 --gradient-accumulation-steps 1 --lr 0.00533 \
  --prefetch-workers 12 --prefetch-depth 16 --scaler-init-scale 1024 \
  --out models/v7_stage1_npu2.pth
# ⚠ shell/train_sft_npu_2card.sh 里带 --eval-every 2000 --early-stop 1 --export-onnx 1
#   —— 段 1 下这三项被跳过；要跑纯 SFT 请手写 torchrun 去掉它们。

# --- 4 × 910A：每卡 batch 1000，accum 2 ⇒ 有效 8000，LR 0.00637 ---
export OMP_NUM_THREADS=1 MKL_NUM_THREADS=1
python run.py --sh shell/train_sft_npu_4card_katago_se.sh --v7 1 \
  --batch-size 1000 --gradient-accumulation-steps 2 --lr 0.00637 \
  --prefetch-workers 6 --prefetch-depth 16 --scaler-init-scale 1024 \
  --out models/v7_stage1_npu4.pth

# ============ 段 2：软标签蒸馏（吃段 1 权重，只训 policy）============
# ⚠ **必须配 --soft-only-sampling 1**：软 CE 的 mask=0 的行贡献**恰好 0**
#   （不是退化成 one-hot CE）⇒ 不开的话只有约 1% 的行贡献，policy 项被缩小约 100×。
# ⚠ --soft-index 接的是 data/labels/soft_index.npz（160.5 MB，已生成）。
# ⚠ 权重必须与段 1 同形（都从段 1 的 checkpoint 续），否则 load_state_dict 会形状不匹配。
python run.py --sh shell/train_sft_npu_4card_katago_se.sh --v7 1 \
  --model models/v7_stage1_npu4.pth \
  --soft-index data/labels/soft_index.npz --soft-only-sampling 1 \
  --soft-weight 1.0 --soft-every 1 \
  --batch-size 1000 --gradient-accumulation-steps 2 --lr 0.00637 \
  --prefetch-workers 6 --prefetch-depth 16 --scaler-init-scale 1024 \
  --out models/v7_stage2_npu4.pth


# ============ 段 3：RL 自对弈（⛔ 单卡 910A）============
# 🔴 **硬件只有一张 910A，且没有多卡 RL 脚本**（shell/ 下只有 train_rl_npu_1card.sh）。
#    ⇒ SFT 用 2/4 卡、RL 用 1 卡，**两者不能同时跑**（同一批硬件），必须串行排期。
#    selfplay_train.py 有 `--ddp 1`（默认 0）+ torchrun 入口，但**从未在多卡跑过**，
#    没有多卡 RL 脚本也没有多卡 RL 参数依据 ⇒ 本节只给单卡。
# 🔴 **RL 是 CPU 瓶颈**：CPU 预算 = parallel_games × mcts_threads。24 核不分片给 RL
#    ⇒ 8 × 3 = 24（与 train_rl_npu_1card.sh 一致）。
# ⚠ --sims 48 是**刻意下调**（默认 400），19 路太慢。想提速先加 --sims，别先加 parallel-games
#   （MCTS 线程是效率损失，局数是并行度损失）。
# ⚠ --model 必须指向**真实存在**的 checkpoint —— train_rl_npu_1card.sh 里写的
#   models/sft_19x19_v18.pth **不存在**（models/ 只有 sft_19x19_v12.pth）。
# ⚠ MCTS 是通道无关的（mcts.py 从 checkpoint 的 stem 读 in_channels），
#   且 inference.py 已注册 22 通道 builder ⇒ 理论能吃 V7 权重，但**未验证**。
OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 python scripts/selfplay_train.py \
  --device npu --board-size 19 \
  --model models/v7_stage2_npu4.pth --out models/az_v7 \
  --iters 20 --sims 48 --parallel-games 8 --mcts-threads 3 \
  --batch-cap 64 --expand-topk 16 --buffer-size 500 \
  --td 1 --td-steps 3 --use-rollout 1 --leaf-ab-depth 2 --rollout-lambda 0.25 \
  --grad-accum-steps 1 --mcts-vector-backup 1 --lr 1e-3 --swanlab 1
# ⚠ --iters 20 是 1 卡脚本的默认值，**不是实测值** —— 请按实际收敛曲线调。
# ⚠ RL 的 batch/lr **没有定标律可依**（不像 SFT 有 √ 律），这两个数是起点不是实测值。

# 软标签命中率 0       → pos_hash 口径（棋盘+to_play+ko 三样）被动过；⛔ 勿改 src/data/pos_hash.py。
# 全局贴目特征 = 32.5  → foxpro 的 KM[650] 是 ×100 家族（1.62%）；⛔ 别传 --no-komi-outlier-fix。
# ImportError: scipy   → requirements.txt 漏列 scipy，手动装。
# sidecar 匹配率 <100%  → 未匹配的局权重会置 0（不静默填垃圾），看脚本末行的覆盖报告。
